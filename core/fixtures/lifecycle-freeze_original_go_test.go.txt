func TestFreezeOriginalLifecycle(t *testing.T) {
 now := time.Date(2026,11,14,12,0,0,0,time.UTC)
 var fixtures []map[string]any
 for _, st := range []transport.Status{
  {}, {State:transport.StateStopped},
  {State:transport.StateConnected,LocalAddr:"127.0.0.1:25565",LocalPort:25565,Path:transport.PathDirect,LatencyMS:12.5,LatencyKnown:true,GameReachable:"yes",Since:now,ActiveConns:2,BytesFromServer:7},
  {State:transport.StateUnreachable,Message:"No response",Path:transport.PathRelay,LatencyMS:2.6,LatencyKnown:true},
 } {
  fixtures=append(fixtures,map[string]any{"kind":"view","status":st,"output":toConnectionView(&entry{status:st})})
  fixtures=append(fixtures,map[string]any{"kind":"path","status":st,"output":describePath(st)})
 }
 for _, text := range []string{"","no response","connection refused","échec"} {
  fixtures=append(fixtures,map[string]any{"kind":"friendly","text":text,"output":friendly(errors.New(text))})
 }
 for _, state := range []transport.State{transport.StateConnected,transport.StateConnecting,transport.StateReconnecting,transport.StateUnreachable,transport.StateStopped,transport.StateError} {
  fixtures=append(fixtures,map[string]any{"kind":"ended","state":state,"output":tunnelEnded(state)})
 }
 for _, pair := range [][2]transport.State{{transport.StateConnected,transport.StateReconnecting},{transport.StateReconnecting,transport.StateUnreachable},{transport.StateUnreachable,transport.StateConnected},{transport.StateConnected,transport.StateConnected}} {
  m:=&Manager{entries:map[string]*entry{"s":{gen:7,status:transport.Status{State:pair[0]}}},log:diagnostics.NewLog(20,"")}
  m.onStatus("s",7,"World",transport.Status{State:pair[1]})
  var logs []map[string]string
  for _,e:=range m.log.Recent(0) {logs=append(logs,map[string]string{"kind":string(e.Kind),"message":e.Message})}
  fixtures=append(fixtures,map[string]any{"kind":"status","previous":pair[0],"status":transport.Status{State:pair[1]},"logs":logs,"output":toConnectionView(m.entries["s"])})
 }
 b,err:=json.MarshalIndent(fixtures,"","  ");if err!=nil {t.Fatal(err)}
 if err=os.WriteFile("testdata/lifecycle-go-6c89f64.json",append(b,'\n'),0600);err!=nil {t.Fatal(err)}
}
